"?role=business" : ""); return false; }; window.doSignup = async function () { const msgEl = document.getElementById("signupMsg"); msgEl.classList.remove("show", "success"); const name = document.getElementById("f-name").value.trim(); const company = document.getElementById("f-company").value.trim(); const email = document.getElementById("f-email").value.trim(); const password = document.getElementById("f-password").value; if (!name || !email || password.length < 8) { msgEl.textContent = "Fill in your name, a valid email, and a password of at least 8 characters."; msgEl.classList.add("show"); return; } if (selectedRole === "business" && !company) { msgEl.textContent = "Enter your company name."; msgEl.classList.add("show"); return; } const btn = document.getElementById("signupBtn"); btn.disabled = true; btn.innerHTML = ' Creating account…'; try { const res = await fetch(API_BASE + "/auth/register", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ email, password, name, role: selectedRole, companyName: selectedRole === "business" ? company : undefined, referralCode: refCode || undefined, }), }); const data = await res.json(); if (!res.ok || !data.success) { msgEl.textContent = data.message || "Something went wrong creating your account."; msgEl.classList.add("show"); btn.disabled = false; btn.textContent = "Create Account"; return; } // Persist the session so every other page in the app can use it. localStorage.setItem("capella_token", data.token); localStorage.setItem("capella_uid", data.uid); localStorage.setItem("capella_role", selectedRole); window.location.href = "capella-activation.html?role=" + selectedRole; } catch (err) { msgEl.textContent = "Could not reach Capella's servers — check your connection and try again."; msgEl.classList.add("show"); btn.disabled = false; btn.textContent = "Create Account"; } };
